// ═══════════════════════════════════════════════════════════════════
// FIS Bob-a-thon — Base Pipeline (Jenkinsfile)
//
// Starting point for the 5-lab workshop. Provisions a Kubernetes
// agent pod with three containers — build-tools (Maven + JDK17),
// oc-tools (OpenShift CLI), and bob (IBM Bob CLI) — sharing a
// workspace-volume emptyDir mounted at /workspace on all three,
// with HOME set to /workspace so Bob can read .bob/custom_modes.yaml
// from the checkout.
//
// This file has a single Checkout stage. Each lab adds one more
// stage beneath it; see labs/LAB<N>_*.md for instructions.
//
// Before running: if your OpenShift project is NOT named `jenkins`,
// update the `bob` container image URL below.
// ═══════════════════════════════════════════════════════════════════

// Per-instance Jira credential routing. Each pipeline self-selects which
// jira-creds-{a..n} secret to mount based on the user number parsed
// from its job name. See setup/JIRA_ACCOUNT_SETUP.md Section 3.3.
//
// Users are paired across 14 Jira accounts:
//   user1/2 -> jira-creds-a, user3/4 -> jira-creds-b, ...,
//   user27/28 -> jira-creds-n. user29+ share user28's instance
//   (jira-creds-n) — we only provisioned 14 Jira accounts.
//
// @NonCPS keeps the regex Matcher object inside this method so it never
// becomes a CPS-serialized local variable (Matcher is not Serializable
// and would crash the pipeline on checkpoint).
@NonCPS
def routeJiraSecret(String jobName) {
    def m = jobName =~ /user0*(\d+)/
    if (!m) return 'jira-creds-n'
    int userNum = m[0][1].toInteger()
    if (userNum < 1) return 'jira-creds-n'
    def letters = ['a','b','c','d','e','f','g','h','i','j','k','l','m','n']
    int letterIdx = Math.min((userNum - 1).intdiv(2), 13)
    return "jira-creds-${letters[letterIdx]}"
}

def jiraSecret = routeJiraSecret(env.JOB_NAME ?: '')

pipeline {
    agent {
        kubernetes {
            yaml """
apiVersion: v1
kind: Pod
spec:
  serviceAccountName: jenkins
  containers:
  - name: build-tools
    image: maven:3.9-eclipse-temurin-17
    command: ['sleep', 'infinity']
    workingDir: /workspace
    volumeMounts:
    - name: workspace-volume
      mountPath: /workspace
    env:
    - name: HOME
      value: /workspace
  - name: oc-tools
    image: quay.io/openshift/origin-cli:latest
    command: ['sleep', 'infinity']
    workingDir: /workspace
    volumeMounts:
    - name: workspace-volume
      mountPath: /workspace
    env:
    - name: HOME
      value: /workspace
  - name: lint-tools
    image: image-registry.openshift-image-registry.svc:5000/jenkins/lint-tools:latest
    command: ['sleep', 'infinity']
    workingDir: /workspace
    volumeMounts:
    - name: workspace-volume
      mountPath: /workspace
    env:
    - name: HOME
      value: /workspace
  - name: bob
    image: image-registry.openshift-image-registry.svc:5000/jenkins/bob-cli:latest
    command: ['sleep', 'infinity']
    workingDir: /workspace
    volumeMounts:
    - name: workspace-volume
      mountPath: /workspace
    env:
    - name: BOBSHELL_API_KEY
      valueFrom:
        secretKeyRef:
          name: bob-cli-credentials
          key: BOBSHELL_API_KEY
    - name: BOB_ACCEPT_LICENSE
      value: "true"
    - name: HOME
      value: /workspace
    - name: JIRA_URL
      valueFrom:
        secretKeyRef:
          name: ${jiraSecret}
          key: JIRA_URL
    - name: JIRA_USERNAME
      valueFrom:
        secretKeyRef:
          name: ${jiraSecret}
          key: JIRA_USERNAME
    - name: JIRA_API_TOKEN
      valueFrom:
        secretKeyRef:
          name: ${jiraSecret}
          key: JIRA_API_TOKEN
    - name: JIRA_PROJECT
      valueFrom:
        secretKeyRef:
          name: ${jiraSecret}
          key: JIRA_PROJECT
  volumes:
  - name: workspace-volume
    emptyDir: {}
"""
            defaultContainer 'build-tools'
        }
    }

    stages {
        stage('Checkout') {
            steps {
                echo "=== Checking out repository ==="
                checkout scm

                echo "=== Verifying workspace contents ==="
                sh '''
                    echo "Current directory: $(pwd)"
                    echo "Workspace contents:"
                    ls -la
                '''
            }
        }

        // ── Lab 1: PR / Git Diff Review ──────────────────────────
        stage('PR Review') {
            options {
                timeout(time: 5, unit: 'MINUTES')
            }
            steps {
                script {
                    echo '════════════════════════════════════════════════════════'
                    echo '  🤖 Bob PR Review Analysis'
                    echo "  Started: ${new Date()}"
                    echo '════════════════════════════════════════════════════════'
                    
                    catchError(buildResult: 'UNSTABLE', stageResult: 'FAILURE') {
                        // Configure git safe directory and compute diff
                        sh '''
                            git config --global --add safe.directory "$WORKSPACE"
                            git diff origin/main...HEAD > git-diff.txt || : > git-diff.txt
                        '''
                        
                        // Build prompt for Bob
                        def prompt = """
Read git-diff.txt and produce a senior developer PR overview.

Focus on:
- What changed (1-2 sentences per notable change)
- Risk level (high/medium/low) for each change
- Watch for: null-safety, concurrency, behavior changes, security surface, performance hot paths, missing tests

Keep it brief - if nothing notable changed, say so in one sentence.
"""
                        
                        // Run Bob analysis
                        def analysis = askBob(prompt, 'pipeline-git-diff-overview')
                        
                        // Display in console
                        echo ''
                        echo analysis
                        echo ''
                        
                        // Save for archiving
                        writeFile file: 'bob-pr-review.md', text: analysis
                    }
                    
                    echo "  Completed: ${new Date()}"
                    echo "  Full report: bob-pr-review.md (archived)"
                    echo '════════════════════════════════════════════════════════'
                }
            }
            post {
                always {
                    archiveArtifacts artifacts: 'bob-pr-review.md',
                                   allowEmptyArchive: true,
                                   fingerprint: true
                }
            }
        }

        // ── Lab 2: Unit Testing ──────────────────────────────────
        stage('Unit Tests') {
            options {
                timeout(time: 10, unit: 'MINUTES')
            }
            steps {
                script {
                    echo '════════════════════════════════════════════════════════'
                    echo '  🧪 Running Unit Tests'
                    echo "  Started: ${new Date()}"
                    echo '════════════════════════════════════════════════════════'
                    
                    // Run tests and capture result (continue even if tests fail)
                    def testResult = sh(
                        script: 'mvn -f order-service/pom.xml test',
                        returnStatus: true
                    )
                    
                    // Publish JUnit test results
                    junit testResults: 'order-service/target/surefire-reports/*.xml',
                          allowEmptyResults: true
                    
                    // If tests failed, analyze with Bob
                    if (testResult != 0) {
                        echo ''
                        echo '  ⚠️  Tests failed - analyzing with Bob...'
                        echo ''
                        
                        catchError(buildResult: 'UNSTABLE', stageResult: 'FAILURE') {
                            // Read test failure reports
                            def reports = sh(
                                script: 'find order-service/target/surefire-reports -name "*.txt" -exec cat {} + || true',
                                returnStdout: true
                            ).trim()
                            
                            // Build prompt for Bob
                            def prompt = """
Analyze the following test failures from order-service/target/surefire-reports/:

${reports}

For each failed test:
- One sentence on what failed and where
- Likely root cause (assertion mismatch, NPE, mock misconfiguration, timeout, wrong exception type)
- Suggested fix with a short code snippet
- Group related failures if they share a root cause

Read the test source files under order-service/src/test/ and production code under order-service/src/main/ as needed.

Output in plain text with sections: Summary, Failures, Suggested Fixes.
"""
                            
                            // Run Bob analysis
                            def analysis = askBob(prompt, 'pipeline-test-failure-analyzer')
                            
                            // Display in console
                            echo ''
                            echo '════════════════════════════════════════════════════════'
                            echo '  🔬 Bob Test Failure Analysis'
                            echo '════════════════════════════════════════════════════════'
                            echo analysis
                            echo '════════════════════════════════════════════════════════'
                            echo ''
                            
                            // Save for archiving
                            writeFile file: 'bob-test-analysis.md', text: analysis
                        }
                        
                        // Mark build as unstable
                        currentBuild.result = 'UNSTABLE'
                    } else {
                        echo ''
                        echo '  ✅ All tests passed'
                    }
                    
                    echo ''
                    echo "  Completed: ${new Date()}"
                    if (testResult != 0) {
                        echo "  Test analysis: bob-test-analysis.md (archived)"
                    }
                    echo '════════════════════════════════════════════════════════'
                }
            }
            post {
                always {
                    archiveArtifacts artifacts: 'bob-test-analysis.md',
                                   allowEmptyArchive: true,
                                   fingerprint: true
                }
            }
        }

        // ── Lab 3: Security Analysis ─────────────────────────────
        stage('Security Analysis') {
            options {
                timeout(time: 15, unit: 'MINUTES')
            }
            environment {
                SONAR_TOKEN = credentials('sonarqube-token')
                SONAR_HOST_URL = 'https://sonarqube-sonarqube.apps.itz-vl5o43.infra01-lb.wdc04.techzone.ibm.com'
                PROJECT_KEY = "order-service-${env.USER}-${env.BUILD_NUMBER}"
            }
            steps {
                script {
                    echo '════════════════════════════════════════════════════════'
                    echo '  🔒 Multi-Layer Security Analysis'
                    echo "  Started: ${new Date()}"
                    echo '════════════════════════════════════════════════════════'
                    
                    // Initialize results map with serializable primitives only
                    def securityResults = [
                        sonarqube: [:],
                        grype: [:],
                        riskLevel: '',
                        riskScore: 0,
                        deploymentDecision: ''
                    ]
                    
                    // ═══════════════════════════════════════════════════════
                    // Phase 1: SonarQube Security Scanning
                    // ═══════════════════════════════════════════════════════
                    catchError(buildResult: 'UNSTABLE', stageResult: 'FAILURE') {
                        echo ''
                        echo '📊 Phase 1: SonarQube Security Scanning'
                        echo '─────────────────────────────────────────────────────'
                        
                        // Run SonarQube scan
                        sh """
                            cd order-service
                            mvn clean test-compile sonar:sonar \
                                -Dsonar.projectKey=${PROJECT_KEY} \
                                -Dsonar.projectName="Order Service ${env.USER}" \
                                -Dsonar.host.url=${SONAR_HOST_URL} \
                                -Dsonar.token=${SONAR_TOKEN} \
                                -DskipTests
                        """
                        
                        echo '✅ SonarQube scan completed'
                        echo ''
                        echo '⏳ Phase 1a: Waiting for SonarQube task completion...'
                        
                        // Phase 1a: Wait for task completion (12 retries, 10 seconds)
                        def taskCompleted = false
                        def taskStatus = ''
                        for (int i = 0; i < 12; i++) {
                            try {
                                taskStatus = sh(
                                    script: """
                                        curl -s -u demo:Demo123lab123@ \
                                            "${SONAR_HOST_URL}/api/ce/component?component=${PROJECT_KEY}" \
                                            | grep -o '"status":"[^"]*"' | head -1 | cut -d'"' -f4
                                    """,
                                    returnStdout: true
                                ).trim()
                                
                                echo "  Attempt ${i+1}/12: Task status = ${taskStatus}"
                                
                                if (taskStatus == 'SUCCESS') {
                                    taskCompleted = true
                                    echo '✅ SonarQube task completed successfully'
                                    break
                                } else if (taskStatus == 'FAILED') {
                                    echo '❌ SonarQube task failed'
                                    break
                                }
                                
                                if (i < 11) {
                                    sleep(10)
                                }
                            } catch (Exception e) {
                                echo "  ⚠️  Error checking task status: ${e.message}"
                                if (i < 11) {
                                    sleep(10)
                                }
                            }
                        }
                        
                        if (!taskCompleted) {
                            echo '⚠️  Task did not complete within timeout, proceeding anyway...'
                        }
                        
                        echo ''
                        echo '📥 Phase 1b: Fetching SonarQube metrics...'
                        
                        // Phase 1b: Fetch metrics (6 retries, 5 seconds)
                        def metricsJson = ''
                        def metricsFetched = false
                        for (int i = 0; i < 6; i++) {
                            try {
                                metricsJson = sh(
                                    script: """
                                        curl -s -u demo:Demo123lab123@ \
                                            "${SONAR_HOST_URL}/api/measures/component?component=${PROJECT_KEY}&metricKeys=bugs,vulnerabilities,code_smells,security_hotspots,security_rating,reliability_rating,security_review_rating"
                                    """,
                                    returnStdout: true
                                ).trim()
                                
                                if (metricsJson && metricsJson.contains('"component"')) {
                                    metricsFetched = true
                                    echo "  ✅ Metrics fetched successfully (attempt ${i+1}/6)"
                                    break
                                }
                                
                                if (i < 5) {
                                    echo "  ⏳ Waiting for metrics... (attempt ${i+1}/6)"
                                    sleep(5)
                                }
                            } catch (Exception e) {
                                echo "  ⚠️  Error fetching metrics: ${e.message}"
                                if (i < 5) {
                                    sleep(5)
                                }
                            }
                        }
                        
                        if (metricsFetched) {
                            // Parse JSON using Groovy JsonSlurper
                            def sonarData = null
                            try {
                                sonarData = new groovy.json.JsonSlurper().parseText(metricsJson)
                                
                                if (sonarData?.component?.measures && sonarData.component.measures.size() > 0) {
                                    // Store as String primitives to avoid serialization issues
                                    sonarData.component.measures.each { measure ->
                                        securityResults.sonarqube[measure.metric] = measure.value ?: '0'
                                    }
                                    
                                    echo ''
                                    echo '📊 SonarQube Results:'
                                    echo "  Bugs: ${securityResults.sonarqube.bugs ?: '0'}"
                                    echo "  Vulnerabilities: ${securityResults.sonarqube.vulnerabilities ?: '0'}"
                                    echo "  Code Smells: ${securityResults.sonarqube.code_smells ?: '0'}"
                                    echo "  Security Hotspots: ${securityResults.sonarqube.security_hotspots ?: '0'}"
                                    echo "  Security Rating: ${securityResults.sonarqube.security_rating ?: 'N/A'}"
                                    echo "  Reliability Rating: ${securityResults.sonarqube.reliability_rating ?: 'N/A'}"
                                } else {
                                    echo '⚠️  No measures found in SonarQube response'
                                }
                            } catch (Exception e) {
                                echo "⚠️  Error parsing SonarQube metrics: ${e.message}"
                            } finally {
                                // Clear JSON object to avoid serialization issues
                                sonarData = null
                            }
                        } else {
                            echo '⚠️  Could not fetch SonarQube metrics within timeout'
                        }
                    }
                    
                    // ═══════════════════════════════════════════════════════
                    // Phase 2: Grype Vulnerability Scanning
                    // ═══════════════════════════════════════════════════════
                    catchError(buildResult: 'UNSTABLE', stageResult: 'FAILURE') {
                        echo ''
                        echo '🔍 Phase 2: Grype Vulnerability Scanning'
                        echo '─────────────────────────────────────────────────────'
                        
                        container('build-tools') {
                            // Install Grype on-demand
                            echo '📦 Installing Grype...'
                            sh '''
                                curl -sSfL https://raw.githubusercontent.com/anchore/grype/main/install.sh | sh -s -- -b /usr/local/bin
                                grype version
                            '''
                            
                            // Run Grype scan
                            echo '🔎 Running Grype scan...'
                            sh '''
                                cd order-service
                                grype dir:. --scope all-layers -o json > grype-report.json || true
                            '''
                            
                            // Parse results using grep and wc (no jq dependency)
                            echo '📊 Parsing Grype results...'
                            def criticalCount = sh(
                                script: 'grep -o \'"severity":"Critical"\' order-service/grype-report.json | wc -l || echo 0',
                                returnStdout: true
                            ).trim().toInteger()
                            
                            def highCount = sh(
                                script: 'grep -o \'"severity":"High"\' order-service/grype-report.json | wc -l || echo 0',
                                returnStdout: true
                            ).trim().toInteger()
                            
                            def mediumCount = sh(
                                script: 'grep -o \'"severity":"Medium"\' order-service/grype-report.json | wc -l || echo 0',
                                returnStdout: true
                            ).trim().toInteger()
                            
                            // Store as Integer primitives
                            securityResults.grype.critical = criticalCount
                            securityResults.grype.high = highCount
                            securityResults.grype.medium = mediumCount
                            securityResults.grype.total = criticalCount + highCount + mediumCount
                            
                            echo ''
                            echo '📊 Grype Results:'
                            echo "  Critical: ${criticalCount}"
                            echo "  High: ${highCount}"
                            echo "  Medium: ${mediumCount}"
                            echo "  Total: ${securityResults.grype.total}"
                        }
                    }
                    
                    // ═══════════════════════════════════════════════════════
                    // Phase 3: CVE Analysis with Bob (conditional)
                    // ═══════════════════════════════════════════════════════
                    def totalVulns = (securityResults.grype.total ?: 0)
                    if (totalVulns > 0) {
                        catchError(buildResult: 'UNSTABLE', stageResult: 'FAILURE') {
                            echo ''
                            echo '🤖 Phase 3: CVE Analysis with Bob'
                            echo '─────────────────────────────────────────────────────'
                            
                            def cvePrompt = """
Analyze the security findings from our multi-layer security scan:

**SonarQube Results:**
- Bugs: ${securityResults.sonarqube.bugs ?: '0'}
- Vulnerabilities: ${securityResults.sonarqube.vulnerabilities ?: '0'}
- Code Smells: ${securityResults.sonarqube.code_smells ?: '0'}
- Security Hotspots: ${securityResults.sonarqube.security_hotspots ?: '0'}
- Security Rating: ${securityResults.sonarqube.security_rating ?: 'N/A'}

**Grype Vulnerability Scan:**
- Critical CVEs: ${securityResults.grype.critical}
- High CVEs: ${securityResults.grype.high}
- Medium CVEs: ${securityResults.grype.medium}
- Total Vulnerabilities: ${totalVulns}

Please provide:
1. **Risk Assessment**: Overall security posture (1-2 sentences)
2. **Top 3 Vulnerabilities**: Most critical issues to address
3. **Remediation Steps**: Specific actions to fix top issues
4. **Deployment Recommendation**: One of:
   - BLOCK: Critical issues, do not deploy
   - WARN: Significant issues, manual review required
   - PROCEED: Minor issues, safe to deploy with approval

Format your response clearly with these sections.
"""
                            
                            def cveAnalysis = askBob(cvePrompt, null)
                            
                            echo ''
                            echo '📋 Bob CVE Analysis:'
                            echo cveAnalysis
                            
                            // Save analysis
                            writeFile file: 'bob-cve-analysis.txt', text: cveAnalysis
                            
                            // Extract deployment recommendation
                            if (cveAnalysis.contains('BLOCK')) {
                                securityResults.deploymentDecision = 'BLOCK'
                            } else if (cveAnalysis.contains('WARN')) {
                                securityResults.deploymentDecision = 'WARN'
                            } else if (cveAnalysis.contains('PROCEED')) {
                                securityResults.deploymentDecision = 'PROCEED'
                            } else {
                                securityResults.deploymentDecision = 'WARN'
                            }
                        }
                    } else {
                        echo ''
                        echo '✅ Phase 3: No vulnerabilities found, skipping CVE analysis'
                        securityResults.deploymentDecision = 'PROCEED'
                    }
                    
                    // ═══════════════════════════════════════════════════════
                    // Phase 4: Risk Level Calculation
                    // ═══════════════════════════════════════════════════════
                    echo ''
                    echo '📊 Phase 4: Risk Level Calculation'
                    echo '─────────────────────────────────────────────────────'
                    
                    // Parse SonarQube String values to Integer safely
                    def bugs = 0
                    def vulnerabilities = 0
                    def hotspots = 0
                    
                    try {
                        bugs = Integer.parseInt(securityResults.sonarqube.bugs ?: '0')
                    } catch (Exception e) {
                        bugs = 0
                    }
                    
                    try {
                        vulnerabilities = Integer.parseInt(securityResults.sonarqube.vulnerabilities ?: '0')
                    } catch (Exception e) {
                        vulnerabilities = 0
                    }
                    
                    try {
                        hotspots = Integer.parseInt(securityResults.sonarqube.security_hotspots ?: '0')
                    } catch (Exception e) {
                        hotspots = 0
                    }
                    
                    // Calculate risk score
                    def riskScore = (vulnerabilities * 10) +
                                   (hotspots * 5) +
                                   (bugs * 2) +
                                   ((securityResults.grype.critical ?: 0) * 20) +
                                   ((securityResults.grype.high ?: 0) * 10)
                    
                    securityResults.riskScore = riskScore
                    
                    // Determine risk level
                    if (riskScore >= 50) {
                        securityResults.riskLevel = 'CRITICAL'
                    } else if (riskScore >= 30) {
                        securityResults.riskLevel = 'HIGH'
                    } else if (riskScore >= 10) {
                        securityResults.riskLevel = 'MEDIUM'
                    } else {
                        securityResults.riskLevel = 'LOW'
                    }
                    
                    echo "  Risk Score: ${riskScore}"
                    echo "  Risk Level: ${securityResults.riskLevel}"
                    
                    // ═══════════════════════════════════════════════════════
                    // Phase 5: Deployment Decision
                    // ═══════════════════════════════════════════════════════
                    echo ''
                    echo '🚦 Phase 5: Deployment Decision'
                    echo '─────────────────────────────────────────────────────'
                    
                    if (securityResults.riskLevel == 'CRITICAL') {
                        echo '🔴 CRITICAL: Blocking deployment due to critical security issues'
                        currentBuild.result = 'UNSTABLE'
                        securityResults.deploymentDecision = 'BLOCK'
                    } else if (securityResults.riskLevel == 'HIGH') {
                        echo '🟡 HIGH: Manual review recommended before deployment'
                        securityResults.deploymentDecision = 'WARN'
                    } else {
                        echo '🟢 MEDIUM/LOW: Proceed with deployment approval'
                        if (!securityResults.deploymentDecision) {
                            securityResults.deploymentDecision = 'PROCEED'
                        }
                    }
                    
                    echo "  Decision: ${securityResults.deploymentDecision}"
                    
                    // ═══════════════════════════════════════════════════════
                    // Phase 6: Consolidated Report Generation
                    // ═══════════════════════════════════════════════════════
                    echo ''
                    echo '📝 Phase 6: Generating Consolidated Report'
                    echo '─────────────────────────────────────────────────────'
                    
                    def report = """
# Security Analysis Report

**Build:** ${env.BUILD_NUMBER}
**Date:** ${new Date()}
**Project:** ${PROJECT_KEY}

---

## Executive Summary

**Risk Level:** ${securityResults.riskLevel}
**Risk Score:** ${securityResults.riskScore}
**Deployment Decision:** ${securityResults.deploymentDecision}

---

## SonarQube Results

| Metric | Count |
|--------|-------|
| Bugs | ${securityResults.sonarqube.bugs ?: '0'} |
| Vulnerabilities | ${securityResults.sonarqube.vulnerabilities ?: '0'} |
| Code Smells | ${securityResults.sonarqube.code_smells ?: '0'} |
| Security Hotspots | ${securityResults.sonarqube.security_hotspots ?: '0'} |
| Security Rating | ${securityResults.sonarqube.security_rating ?: 'N/A'} |
| Reliability Rating | ${securityResults.sonarqube.reliability_rating ?: 'N/A'} |

**Dashboard:** ${SONAR_HOST_URL}/dashboard?id=${PROJECT_KEY}

---

## Grype Vulnerability Scan

| Severity | Count |
|----------|-------|
| Critical | ${securityResults.grype.critical ?: 0} |
| High | ${securityResults.grype.high ?: 0} |
| Medium | ${securityResults.grype.medium ?: 0} |
| **Total** | **${securityResults.grype.total ?: 0}** |

---

## CVE Analysis

${totalVulns > 0 ? 'See bob-cve-analysis.txt for detailed CVE analysis and remediation recommendations.' : 'No vulnerabilities detected - CVE analysis skipped.'}

---

## Deployment Recommendation

**${securityResults.deploymentDecision}**

${securityResults.riskLevel == 'CRITICAL' ? '🔴 **BLOCKED**: Critical security issues must be resolved before deployment.' : ''}
${securityResults.riskLevel == 'HIGH' ? '🟡 **WARNING**: Manual security review required before proceeding.' : ''}
${securityResults.riskLevel == 'MEDIUM' || securityResults.riskLevel == 'LOW' ? '🟢 **PROCEED**: Security posture acceptable with standard approval process.' : ''}

---

**Report Generated:** ${new Date()}
"""
                    
                    writeFile file: 'security-analysis-report.md', text: report
                    
                    echo '✅ Consolidated report generated: security-analysis-report.md'
                    echo ''
                    echo '════════════════════════════════════════════════════════'
                    echo '  🔒 Security Analysis Complete'
                    echo "  Risk Level: ${securityResults.riskLevel}"
                    echo "  Decision: ${securityResults.deploymentDecision}"
                    echo "  Completed: ${new Date()}"
                    echo '════════════════════════════════════════════════════════'
                }
            }
            post {
                always {
                    archiveArtifacts artifacts: 'security-analysis-report.md,bob-cve-analysis.txt,order-service/grype-report.json',
                                   allowEmptyArchive: true,
                                   fingerprint: true
                }
            }
        }
        // ══════════════════════════════════════════════════════════
        // Lab 4: Run Linters
        // ══════════════════════════════════════════════════════════
        stage('Run Linters') {
            options {

// ── Helper: ask Bob, optionally with a specific custom mode ───────────────────
// Writes the prompt to a tempfile in the shared workspace and runs `bob` in
// the bob container, adding `--chat-mode <slug>` only when a mode is provided.
// Returns the analysis as a string. Using a tempfile (instead of inlining the
// prompt on the command line) avoids shell-escaping issues when the prompt
// contains quotes, backticks, or newlines — common with diffs.
def askBob(String prompt, String mode = null) {
    container('bob') {
        def promptFile = ".bob-prompt-${System.currentTimeMillis()}.txt"
        writeFile file: promptFile, text: prompt

        def modeFlag = mode ? "--chat-mode ${mode}" : ""
        def analysis = sh(
            script: """bob ${modeFlag} -p "\$(cat ${promptFile})" --hide-intermediary-output""",
            returnStdout: true
        ).trim()

        sh "rm -f ${promptFile}"
        return analysis
    }
}
